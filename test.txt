int lexer(FILE *target)
{
    size_t filesize = get_filesize(target); 
    
    char *file = malloc(filesize + 1);
    if (file == NULL) 
    {
        return -1; 
    }
    size_t bytes_read = fread(file, 1, filesize, target);
    file[bytes_read] = '\0';


    if(tokenize(file, filesize + 1) = false)
    {
        return 0;
    }

    free(file);
    return 1;
}

void scan_operator(char *file, int *i, size_t filesize, terminate terminator)
{
    char buffer[TOKEN_LITERAL_SIZE];
    int index = 0;
    bool brk = false;
    while (true)
    {
        buffer[index++] = file[*i];
        *i++;
        switch (terminator)
        {
            case STRING:
                if (buffer[index] == '"')
                {
                    brk = true;
                }
                break;
            case PUNCT: /* NOT FINISHED FIX //////////////////////////////////////////////////////////*/
                if (i + 1 < filesize && !match(file[*i], file[*i + 1]))
                {
                    brk = true;
                }
                break;
            case CHARS:
                if (!isalpha(buffer[index]) && buffer[index] != '_')
                {
                    brk = true;
                }
                break;
            case NUMBER:
                if (!isdigit(buffer[index]))
                {
                    brk = true;
                }
                break;
        }

        if (brk)
        {
            buffer[index] = '\0';
            break;
        }
    }


}
